Monitoramento de Motores Industriais

Sistema de validação de telemetria, análise de limites operacionais e plano de manutenção preventiva.

Repositório GitHub

1. Explicação Técnica do Sistema

O software foi desenvolvido em Python puro utilizando estruturas de dados nativas (listas e dicionários) e programação modular para processar leituras de sensores industriais sem dependências externas. O fluxo de execução divide-se em quatro etapas principais:

01

Sanitização e Validação

As funções validar_leitura() e obter_leituras_validas() verificam a integridade do dicionário de cada motor, garantindo que todos os campos obrigatórios existam e descartando valores nulos ou fisicamente inválidos (negativos).

02

Motor de Regras e Alertas

A função analisar_alertas() compara as métricas coletadas contra o dicionário de constantes LIMITES. A criticidade é classificada dinamicamente conforme a quantidade de desvios simultâneos: 0 alertas (NORMAL), 1 alerta (ATENÇÃO) e 2 ou mais alertas (CRÍTICO).

03

Consolidação Estatística

calcular_estatisticas() agrega os dados da planta em tempo real, calculando temperatura média, corrente média, potência ativa total instalada (kW) e identificando o ativo com maior carga térmica.

04

Manutenção Preventiva

gerar_plano_manutencao() categoriza os equipamentos por prioridade de intervenção operacional: parada imediata, inspeção programada ou operação estável.

2. Parâmetros Monitorados e Limites Operacionais

Parâmetros configurados na constante global LIMITES para disparo automático de ocorrências:

Parâmetro Chave no Código Unidade Faixa / Limite Seguro Alerta Gerado em Caso de Desvio
Temperatura temperatura °C ≤ 80.0 °C Superaquecimento
Corrente Elétrica corrente A (Ampères) ≤ 18.5 A Sobrecorrente
Vibração Mecânica vibracao mm/s ≤ 4.5 mm/s Vibração elevada
Rotação rpm RPM 1400 a 1800 RPM Rotação fora da faixa

4. Código-Fonte Python Organizado

Estrutura refatorada com constantes agrupadas, docstrings e ponto de entrada único (main.py).

from datetime import datetime

# ==============================================================================
# CONFIGURAÇÕES E CONSTANTES DO SISTEMA
# ==============================================================================
SISTEMA_NOME = "Monitoramento de Motores Industriais"
VERSAO = "1.0.0"

LIMITES = {
    "temperatura_max": 80.0,
    "corrente_max": 18.5,
    "vibracao_max": 4.5,
    "rpm_min": 1400,
    "rpm_max": 1800,
}

LEITURAS_MOTORES = [
    {"id": "MTR-01", "setor": "Estamparia", "temperatura": 68.4, "corrente": 14.2, "vibracao": 2.1, "rpm": 1750, "potencia_kw": 7.5},
    {"id": "MTR-02", "setor": "Usinagem",   "temperatura": 84.7, "corrente": 19.3, "vibracao": 5.2, "rpm": 1690, "potencia_kw": 11.0},
    {"id": "MTR-03", "setor": "Bobinagem",  "temperatura": 76.2, "corrente": 17.8, "vibracao": 4.7, "rpm": 1520, "potencia_kw": 5.5},
    {"id": "MTR-04", "setor": "Pintura",    "temperatura": 61.0, "corrente": 11.5, "vibracao": 1.8, "rpm": 1780, "potencia_kw": 7.5},
]


# ==============================================================================
# VALIDAÇÃO E PROCESSAMENTO DE DADOS
# ==============================================================================
def validar_leitura(motor: dict) -> bool:
    """Verifica a presença dos campos obrigatórios e descarta valores negativos."""
    campos_obrigatorios = ["id", "setor", "temperatura", "corrente", "vibracao", "rpm", "potencia_kw"]
    for campo in campos_obrigatorios:
        if campo not in motor or motor[campo] is None:
            return False
    if motor["temperatura"] < 0 or motor["corrente"] < 0 or motor["vibracao"] < 0 or motor["rpm"] < 0:
        return False
    return True


def obter_leituras_validas(lista_motores: list) -> list:
    """Filtra e retorna apenas os registros de motores válidos."""
    return [m for m in lista_motores if validar_leitura(m)]


def analisar_alertas(motor: dict) -> tuple[str, list]:
    """Compara os parâmetros do motor com os limites e classifica o status."""
    alertas = []
    if motor["temperatura"] > LIMITES["temperatura_max"]:
        alertas.append(f"Superaquecimento ({motor['temperatura']:.1f} C)")
    if motor["corrente"] > LIMITES["corrente_max"]:
        alertas.append(f"Sobrecorrente ({motor['corrente']:.1f} A)")
    if motor["vibracao"] > LIMITES["vibracao_max"]:
        alertas.append(f"Vibracao elevada ({motor['vibracao']:.1f} mm/s)")
    if motor["rpm"] < LIMITES["rpm_min"] or motor["rpm"] > LIMITES["rpm_max"]:
        alertas.append(f"Rotacao fora da faixa ({motor['rpm']} RPM)")

    if len(alertas) == 0:
        status = "NORMAL"
    elif len(alertas) == 1:
        status = "ATENCAO"
    else:
        status = "CRITICO"

    return status, alertas


def calcular_estatisticas(motores: list) -> dict:
    """Calcula médias operacionais, potência total e identifica o maior aquecimento."""
    total = len(motores)
    if total == 0:
        return {}

    temp_media = sum(m["temperatura"] for m in motores) / total
    corrente_media = sum(m["corrente"] for m in motores) / total
    potencia_total = sum(m["potencia_kw"] for m in motores)
    maior_temp = max(motores, key=lambda m: m["temperatura"])

    return {
        "total_motores": total,
        "temperatura_media": round(temp_media, 2),
        "corrente_media": round(corrente_media, 2),
        "potencia_total_kw": round(potencia_total, 2),
        "motor_mais_quente": maior_temp["id"],
    }


def gerar_plano_manutencao(motores: list) -> dict:
    """Agrupa os motores por criticidade para tomada de decisão preventiva."""
    plano = {"CRITICO": [], "ATENCAO": [], "NORMAL": []}
    for motor in motores:
        status, alertas = analisar_alertas(motor)
        plano[status].append({
            "id": motor["id"],
            "setor": motor["setor"],
            "ocorrencias": len(alertas)
        })
    return plano


# ==============================================================================
# INTERFACE DE SAÍDA (CONSOLE)
# ==============================================================================
def exibir_painel(motores: list) -> None:
    print("=" * 64)
    print(f"{SISTEMA_NOME} - v{VERSAO}")
    print(f"Data/Hora da leitura: {datetime.now().strftime('%d/%m/%Y %H:%M:%S')}")
    print("=" * 64)

    for motor in motores:
        status, alertas = analisar_alertas(motor)
        print(
            f"[{status:<7}] {motor['id']} ({motor['setor']}) | "
            f"Temp: {motor['temperatura']} C | Corr: {motor['corrente']} A | "
            f"Vib: {motor['vibracao']} mm/s | RPM: {motor['rpm']}"
        )
        for alerta in alertas:
            print(f"  -> Alerta: {alerta}")

    stats = calcular_estatisticas(motores)
    print("-" * 64)
    print(
        f"Resumo: {stats['total_motores']} motores | "
        f"Temp. media: {stats['temperatura_media']} C | "
        f"Corr. media: {stats['corrente_media']} A | "
        f"Potencia total: {stats['potencia_total_kw']} kW | "
        f"Maior temp.: {stats['motor_mais_quente']}"
    )
    print("=" * 64)


def exibir_manutencao(motores: list) -> None:
    plano = gerar_plano_manutencao(motores)
    print("\nPLANO DE MANUTENCAO PREVENTIVA:")
    print(f" - Parada imediata (CRITICO): {[m['id'] for m in plano['CRITICO']]}")
    print(f" - Inspecao programada (ATENCAO): {[m['id'] for m in plano['ATENCAO']]}")
    print(f" - Operacao estavel (NORMAL): {[m['id'] for m in plano['NORMAL']]}")


if __name__ == "__main__":
    motores_validos = obter_leituras_validas(LEITURAS_MOTORES)
    exibir_painel(motores_validos)
    exibir_manutencao(motores_validos)
